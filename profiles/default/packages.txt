# snapd - needed for the snap extras.txt entry below (ghostty only,
# since 2026-09-12 - yazi moved to github-release, see extras.txt).
# Confirmed in apt's index (Pop!_OS/Debian/Ubuntu). Not in Arch/pacman's
# official repos at all (AUR-only) - but this doesn't matter on pacman:
# ghostty has a real native pacman package (`extra` repo), so the snap
# extras.txt entry below routes through it directly there (see
# _GLB_EXTRA_NATIVE_OVERRIDES in lib/extras.sh) and snapd itself is
# skipped outright on pacman via _GLB_PACKAGE_SKIP in lib/package.sh -
# confirmed 2026-08-16, no more manual-step pause on Arch-family
# machines. Confirmed real gap on zypper too (2026-08-13, openSUSE
# Tumbleweed VM, real restore of the server profile): `zypper info
# snapd` reports "package 'snapd' not found" in the default repos -
# but ghostty has a real native zypper package too (`repo-oss`), so
# this is a non-issue there as well; snapd is simply never needed on
# openSUSE. dnf unverified for snapd itself, and ghostty falls through
# to snap there (Fedora packages it only via COPR) - confirmed working
# 2026-09-12 once `_glb_ensure_snap_dir` created the `/snap` symlink
# Fedora's snapd package doesn't set up on its own (see lib/extras.sh).
#
# Linux Mint (apt) is its own distinct case, confirmed 2026-08-25 on a
# fresh Mint VM: `apt install snapd` fails with "has no installation
# candidate" even though the package is genuinely in Mint's index -
# Mint ships /etc/apt/preferences.d/nosnap.pref, pinning snapd to
# priority -1 by deliberate policy (not a repo gap like pacman/zypper
# above). Not auto-skipped like snapd:pacman, since the user has a
# real fix (delete that file) rather than no path at all - see
# _GLB_PACKAGE_MANUAL_HINT in lib/package.sh, which now surfaces this
# explanation (and the fix command) at the manual-step pause instead
# of just re-showing a `sudo apt install -y snapd` that would fail
# identically every time. ghostty below fails as a direct consequence
# on Mint until snapd is manually unblocked.
snapd

# CPU architecture info banner (James Tigert / kz6fittycent), pairs
# with fastfetch (now a `github-release` extra - see extras.txt). A
# real native package on all four managers - no per-distro override
# needed, name is "cpufetch" everywhere.
# Confirmed current (1.07, matching upstream) on Fedora 44+ (dnf),
# Arch's `extra` repo (pacman), and openSUSE Tumbleweed (zypper).
# Debian stable/trixie lags one point release behind (1.06-2) - a
# real but minor gap, not worth a workaround; same tolerance as every
# other "not quite latest" package GLB installs via the system package
# manager. Originally added via the `snap` method (2026-08-17) before
# checking whether a native package existed on every distro - switched
# to a plain package once it turned out one does, avoiding an
# unnecessary snapd dependency and the manual-step pause that would
# have caused where snapd itself is a known gap (see snapd's own
# comment above).
cpufetch

# fastfetch (system-info banner) lives in extras.txt now, not here.
# It's not in apt's index on GLB's fresh-VM targets (Pop!_OS 24.04,
# Linux Mint 22.x - "has no installation candidate", confirmed
# 2026-08-07 and again on a fresh Mint VM 2026-08-25), so a bare
# packages.txt line forced a manual-step pause on every apt restore.
# `github-release fastfetch fastfetch-cli/fastfetch` pulls the upstream
# tarball into ~/.local/bin instead; dnf/pacman/zypper still get the
# native package via _GLB_EXTRA_NATIVE_OVERRIDES. See extras.txt.

# neovim (LazyVim config vendored in dotfiles/.config/nvim, real
# LazyVim/starter, public) also lives in extras.txt now, not here. Real
# gap hit on Greg's E7450 (2026-09-06, again 2026-09-11 after a
# reinstall): apt's `neovim` is 0.9.5, LazyVim needs >= 0.11.2, so
# `nvim` refused to load with "requires 0.11.2" - a bare packages.txt
# line silently installs a version too old to work. `github-release-tree
# nvim neovim/neovim nvim-linux-x86_64.tar.gz` extracts the upstream
# release's whole bin/lib/share tree into ~/.local instead (a plain
# `github-release` single-binary grab breaks `nvim`'s syntax/colorscheme
# lookup - confirmed: "E484: Can't open file .../syntax/syntax.vim"
# without its sibling lib/share alongside it). dnf/pacman get the
# native `neovim` package via _GLB_EXTRA_NATIVE_OVERRIDES (both
# confirmed current); zypper isn't verified either way yet, so it also
# gets the upstream tarball for now. See extras.txt.

# Wayland clipboard (`wl-copy`/`wl-paste`) - what makes "yank path" in
# yazi, `"+y` in Neovim, tmux copy-mode-to-clipboard, and `... |
# wl-copy` actually reach the system clipboard on a Wayland session.
# Same package name on apt/dnf/pacman/zypper. X11-only sessions want
# `xclip` or `xsel` instead (not added - `default`'s target is a
# modern Wayland desktop, and Neovim's clipboard provider auto-detects
# whichever is present).
wl-clipboard

# git-delta - a syntax-highlighting pager for `git diff`/`git log

# `git-delta` on all four managers; the binary it installs is `delta`.
# `.gitconfig` uses `delta || less` so an un-restored machine (package
# skipped at a no-TTY restore) still pages with plain `less` instead of
# erroring.
git-delta
