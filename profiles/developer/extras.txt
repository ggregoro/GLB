# GLB profile: developer — extras (installs outside apt/dnf/pacman/zypper)
#
# Format: <method> <name> <spec>
#   curl <name> <url>     - runs `curl -fsSL <url> | bash`. <name> is
#                            checked with `command -v` to detect an
#                            existing install.
#   font <name> <zip-url> - downloads a Nerd Fonts release zip,
#                            extracts it into
#                            ~/.local/share/fonts/<name>, then runs
#                            `fc-cache -f`. <name> becomes the install
#                            directory, so no spaces. Requires unzip
#                            (see packages.txt).
#
# Blank lines and lines starting with '#' are ignored, same as
# packages.txt.

# Code editor - same choice as default.
curl    fresh     https://raw.githubusercontent.com/sinelaw/fresh/refs/heads/master/scripts/install.sh

# mise (language version manager) — picked over per-language tools
# (nvm/pyenv/rustup) as one universal tool/mental model for someone
# newer to managing multiple language runtimes. Official installer is
# designed for exactly this curl-pipe-sh pattern and puts the binary at
# ~/.local/bin/mise (see dotfiles/ for the guarded shell activation).
curl    mise      https://mise.run

# TUI for container management, pairs with the Podman pick above. No
# native package on any of the four managers; its own install script
# (bash-specific, hence the curl-pipe-bash fix above) is the canonical
# distribution method upstream recommends.
curl    lazydocker    https://raw.githubusercontent.com/jesseduffield/lazydocker/master/scripts/install_update_linux.sh

# Nerd Font - required for `eza --icons` (all three shells' ls/ll/la/l
# aliases). See default/extras.txt for the fuller rationale.
font    jetbrains-mono-nerd-font    https://github.com/ryanoasis/nerd-fonts/releases/latest/download/JetBrainsMono.zip
